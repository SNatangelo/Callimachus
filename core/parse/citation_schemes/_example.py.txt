entry to `core/parse/parsers.json`.

NAME = "my-scheme"


def detect(body, sentences, references):
    del body, references
    return float(len(sentences))


def build(sentences, references, window, manuscript_id):
    del sentences, references, window, manuscript_id
    return [], [], [], {}
